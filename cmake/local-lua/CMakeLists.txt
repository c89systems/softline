cmake_minimum_required(VERSION 3.24)
get_filename_component(softline_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
include(../softline_workspace.cmake)
softline_check_workspace("${softline_root}")
if(DEFINED SOFTLINE_LOCAL_LIBRARY_DIR AND NOT SOFTLINE_LOCAL_LIBRARY_DIR STREQUAL "")
  softline_check_generated_path("${softline_root}" "${SOFTLINE_LOCAL_LIBRARY_DIR}")
endif()
project(softline_local_lua LANGUAGES C)
include(../softline_local_runtime.cmake)

set(lua_hash 1c4b4068d67061f2a2231ad2b5422e77acea1487ea9890f6320af614f4373dce)
include(../softline_verified_archive.cmake)
softline_configure_dependency_cache()
if(NOT DEFINED SL_TARGET_ID OR SL_TARGET_ID STREQUAL "")
  message(FATAL_ERROR "Local Lua requires the selected lifecycle target ID")
endif()
softline_verified_archive("lua-5.5.1" "https://www.lua.org/ftp/lua-5.5.1.tar.gz"
  "${lua_hash}" lua-5.5.1.tar.gz archive)
set(lua_cache_root "${softline_root}/.cache")
set(lua_extract "${lua_cache_root}/deps-build/${SL_TARGET_ID}/lua")
set(lua_contract "${lua_cache_root}/dependency-contracts/${SL_TARGET_ID}/lua.txt")
foreach(path "${lua_extract}" "${lua_contract}")
  softline_check_generated_path("${softline_root}" "${path}")
endforeach()
set(lua_contract_value "version=5.5.1\narchive_sha256=${lua_hash}\n")
set(lua_stale FALSE)
if(NOT EXISTS "${lua_contract}")
  set(lua_stale TRUE)
else()
  file(READ "${lua_contract}" lua_existing_contract)
  if(NOT lua_existing_contract STREQUAL lua_contract_value)
    set(lua_stale TRUE)
  endif()
endif()
if(NOT EXISTS "${lua_extract}/lua-5.5.1/src/lua.h")
  set(lua_stale TRUE)
endif()
if(lua_stale)
  file(REMOVE_RECURSE "${lua_extract}")
  file(MAKE_DIRECTORY "${lua_extract}")
  file(ARCHIVE_EXTRACT INPUT "${archive}" DESTINATION "${lua_extract}")
  get_filename_component(lua_contract_dir "${lua_contract}" DIRECTORY)
  file(MAKE_DIRECTORY "${lua_contract_dir}")
  file(WRITE "${lua_contract}" "${lua_contract_value}")
endif()
set(lua_source "${lua_extract}/lua-5.5.1/src")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/lua-source-dir.txt" CONTENT "${lua_source}\n")
file(GLOB lua_sources "${lua_source}/*.c")
list(REMOVE_ITEM lua_sources "${lua_source}/lua.c" "${lua_source}/luac.c")
add_library(local_lua STATIC ${lua_sources})
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_compile_definitions(local_lua PUBLIC LUA_USE_LINUX)
elseif(APPLE)
  target_compile_definitions(local_lua PUBLIC LUA_USE_MACOSX)
elseif(UNIX)
  target_compile_definitions(local_lua PUBLIC LUA_USE_POSIX)
endif()
target_link_libraries(local_lua PUBLIC ${CMAKE_DL_LIBS})
if(UNIX AND NOT APPLE)
  target_link_libraries(local_lua PUBLIC m)
endif()
add_executable(lua "${lua_source}/lua.c")
target_link_libraries(lua PRIVATE local_lua)
if(APPLE)
  # LUA_USE_MACOSX enables the upstream interpreter's readline integration.
  target_link_libraries(lua PRIVATE readline)
endif()
set_target_properties(lua PROPERTIES ENABLE_EXPORTS ON)
set_property(TARGET lua APPEND PROPERTY BUILD_RPATH "${SOFTLINE_LOCAL_LIBRARY_DIR}")
softline_local_runtime(lua)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  get_filename_component(runtime_root "${CMAKE_SYSROOT}/../.." REALPATH)
  file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/runtime-root.txt" CONTENT "${runtime_root}\n")
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  # Lua interaction tests share the interpreter's prepared native configuration.
  # Test-only upstream facade; never installed or linked into libsoftline.
  include(../softline_test_libmdf.cmake)
  set(mdf_lua_hash 3c8f3177c018f8a2e35812b8755cbc01d4892a588cbe9006e63c278867e8947b)
  set(mdf_lua_name "libmdf-lua-${SOFTLINE_TEST_LIBMDF_VERSION}.tar.gz")
  softline_verified_archive("libmdf-lua"
    "https://github.com/c89systems/libmdf/releases/download/v${SOFTLINE_TEST_LIBMDF_VERSION}/${mdf_lua_name}"
    "${mdf_lua_hash}" "${mdf_lua_name}" mdf_lua_archive)
  set(mdf_extract "${CMAKE_CURRENT_BINARY_DIR}/libmdf-source-${mdf_lua_hash}")
  set(mdf_source "${mdf_extract}/libmdf-lua-${SOFTLINE_TEST_LIBMDF_VERSION}")
  if(NOT EXISTS "${mdf_source}/lua/libmdf_core.c")
    file(ARCHIVE_EXTRACT INPUT "${mdf_lua_archive}" DESTINATION "${mdf_extract}")
  endif()
  file(READ "${mdf_source}/VERSION" mdf_source_version)
  string(STRIP "${mdf_source_version}" mdf_source_version)
  if(NOT mdf_source_version STREQUAL SOFTLINE_TEST_LIBMDF_VERSION)
    message(FATAL_ERROR "libmdf Lua source version does not match the pinned SDK")
  endif()
  set(mdf_sdk "${softline_root}/.cache/deps/${SL_TARGET_ID}/libmdf/install")
  if(NOT EXISTS "${mdf_sdk}/lib/pkgconfig/libmdf.pc")
    message(FATAL_ERROR "Prepare the pinned libmdf SDK with make build-debug first")
  endif()
  file(READ "${mdf_sdk}/lib/pkgconfig/libmdf.pc" mdf_pc)
  string(FIND "${mdf_pc}" "Version: ${SOFTLINE_TEST_LIBMDF_VERSION}\n" mdf_version_at)
  if(mdf_version_at LESS 0)
    message(FATAL_ERROR "libmdf Lua tests require the exact pinned SDK version")
  endif()
  find_library(mdf_library NAMES mdf PATHS "${mdf_sdk}/lib" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH REQUIRED)
  set(mdf_modules "${CMAKE_CURRENT_BINARY_DIR}/libmdf")
  configure_file("${mdf_source}/lua/libmdf.lua" "${mdf_modules}/libmdf.lua" COPYONLY)
  add_library(test_libmdf_lua MODULE "${mdf_source}/lua/libmdf_core.c")
  target_include_directories(test_libmdf_lua PRIVATE "${lua_source}" "${mdf_sdk}/include")
  target_link_libraries(test_libmdf_lua PRIVATE "${mdf_library}")
  set_target_properties(test_libmdf_lua PROPERTIES PREFIX "" OUTPUT_NAME core
    LIBRARY_OUTPUT_DIRECTORY "${mdf_modules}/libmdf" BUILD_RPATH "${mdf_sdk}/lib")
  foreach(test console_pty chat_shell chat_policies)
    add_executable(test_${test} "${softline_root}/tests/test_${test}.c")
    target_compile_options(test_${test} PRIVATE
      -std=c89 -Wall -Wextra -Wpedantic -pedantic-errors -Werror)
    target_link_libraries(test_${test} PRIVATE util)
    softline_local_runtime(test_${test})
  endforeach()
endif()
